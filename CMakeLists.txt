cmake_minimum_required(VERSION 3.11)
project(RubiksCube)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(RubiksCube src/main.cpp)

if(EMSCRIPTEN)
    set_target_properties(RubiksCube PROPERTIES SUFFIX ".html")
    target_compile_options(RubiksCube PUBLIC "-sUSE_SDL=2")
    target_link_options(RubiksCube PUBLIC "--shell-file" "${CMAKE_CURRENT_SOURCE_DIR}/shell.html")
    target_link_options(RubiksCube PUBLIC "-sUSE_SDL=2")
    target_link_options(RubiksCube PUBLIC "-sFULL_ES2=1")
    target_link_options(RubiksCube PUBLIC "-sEXPORTED_RUNTIME_METHODS=ccall,cwrap")
    target_link_options(RubiksCube PUBLIC "-sEXPORTED_FUNCTIONS=['_main','_scramble_cube']")
    target_link_options(RubiksCube PUBLIC "-sALLOW_MEMORY_GROWTH=1")
else()
    find_package(SDL2 REQUIRED)
    find_package(OpenGL REQUIRED)
    target_include_directories(RubiksCube PRIVATE ${SDL2_INCLUDE_DIRS} ${OPENGL_INCLUDE_DIR})
    target_link_libraries(RubiksCube PRIVATE ${SDL2_LIBRARIES} ${OPENGL_gl_LIBRARY})
endif()
